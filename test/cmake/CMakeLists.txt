add_test(NAME waveletpp.cmake.install-consumer
	COMMAND ${CMAKE_COMMAND}
		-DTEST_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/consumer
		-DTEST_BINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}/consumer-build
		-DTEST_INSTALL_PREFIX=${CMAKE_CURRENT_BINARY_DIR}/install
		-DTEST_INSTALL_CMAKEDIR=${WAVELETPP_INSTALL_CMAKEDIR}
		-DWAVELETPP_BINARY_DIR=${PROJECT_BINARY_DIR}
		-DTEST_GENERATOR=${CMAKE_GENERATOR}
		-DTEST_GENERATOR_PLATFORM=${CMAKE_GENERATOR_PLATFORM}
		-DTEST_GENERATOR_TOOLSET=${CMAKE_GENERATOR_TOOLSET}
		-DTEST_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}
		-DTEST_CXX_COMPILER=${CMAKE_CXX_COMPILER}
		-DTEST_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}
		-DTEST_BUILD_TYPE=${CMAKE_BUILD_TYPE}
		-DTEST_CONFIGURATION=$<CONFIG>
		-DTEST_VERSION=${PROJECT_VERSION}
		-DTEST_CTEST_COMMAND=${CMAKE_CTEST_COMMAND}
		-P ${CMAKE_CURRENT_SOURCE_DIR}/run_install_consumer.cmake
)
set_tests_properties(waveletpp.cmake.install-consumer PROPERTIES
	LABELS cmake
	RUN_SERIAL TRUE
	TIMEOUT ${WAVELETPP_CMAKE_TEST_TIMEOUT}
)
